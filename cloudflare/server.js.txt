const ADMIN_URL='https://tempahan-mpc-wilayah-utara.groupmpc-0630.chatgpt.site/admin';
const CALENDAR_URL='https://tempahan-mpc-wilayah-utara.groupmpc-0630.chatgpt.site/api/availability';
const json=(value,status=200)=>new Response(JSON.stringify(value),{status,headers:{'Content-Type':'application/json','Cache-Control':'no-store','X-Content-Type-Options':'nosniff'}});
export default {async fetch(request){try{
 const {pathname:path}=new URL(request.url);
 if(['/admin','/admin/','/admin.html'].includes(path))return Response.redirect(ADMIN_URL,302);
 if(path==='/api/session')return json({isAdmin:false,adminUrl:ADMIN_URL});
 if(path==='/api/availability'){
  if(request.method!=='GET')return json({error:'Kemas kini hanya melalui halaman pentadbir asal.',adminUrl:ADMIN_URL},403);
  const upstream=await fetch(CALENDAR_URL,{headers:{Accept:'application/json'},redirect:'error',signal:AbortSignal.timeout(10000)});
  if(!upstream.ok||!upstream.headers.get('content-type')?.includes('application/json'))return json({error:'Kalendar tidak dapat dimuatkan. Sila cuba semula.'},503);
  return new Response(upstream.body,{headers:{'Content-Type':'application/json','Cache-Control':'no-store','X-Content-Type-Options':'nosniff'}});
 }
 if(request.method!=='GET'&&request.method!=='HEAD')return new Response('Kaedah tidak dibenarkan',{status:405});
 const file=FILES[path==='/'?'/index.html':path==='/dashboard'||path==='/dashboard/'?'/dashboard.html':path];
 if(!file)return new Response('Halaman tidak ditemui',{status:404});
 let [body,type,binary]=file;
 const bytes=binary?Uint8Array.from(atob(body),c=>c.charCodeAt(0)):body;
 return new Response(request.method==='HEAD'?null:bytes,{headers:{'Content-Type':type,'Cache-Control':path.startsWith('/assets/')?'public, max-age=86400':'no-cache','X-Content-Type-Options':'nosniff','Referrer-Policy':'strict-origin-when-cross-origin','Content-Security-Policy':"default-src 'self'; img-src 'self' data:; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; font-src https://fonts.gstatic.com; script-src 'self'; object-src 'none'; base-uri 'self'; form-action 'self' mailto:; frame-ancestors 'self'"}});
 }catch(error){console.error(JSON.stringify({event:'request_failed',message:error instanceof Error?error.message:'Unknown error'}));return json({error:'Ralat sementara. Sila cuba semula.'},503);}}};
